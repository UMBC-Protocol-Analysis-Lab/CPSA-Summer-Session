(defprotocol dh diffie-hellman
  (defrole alice
    (vars (a b rndx) (alice bob name) (msg1 msg2 text))
    (trace
      (send (enc "message 1" alice bob (exp (gen) a) (privk alice)))
      (recv (enc alice bob (exp (gen) b) (privk bob)))
      (send (enc "msg 1" msg1 (exp (gen) (mul a b))))
      (recv (enc msg2 (exp (gen) (mul a b)))))
    (uniq-gen a))
  (defrole bob
    (vars (a b rndx) (alice bob name) (msg1 msg2 text))
    (trace
      (recv (enc "message 1" alice bob (exp (gen) a) (privk alice)))
      (send (enc alice bob (exp (gen) b) (privk bob)))
      (recv (enc "msg 1" msg1 (exp (gen) (mul a b))))
      (send (enc msg2 (exp (gen) (mul a b)))))
    (uniq-gen b)
    (absent (b a)))
  (defgenrule neqRl_indx
    (forall ((x indx)) (implies (fact neq x x) (false))))
  (defgenrule neqRl_strd
    (forall ((x strd)) (implies (fact neq x x) (false))))
  (defgenrule neqRl_mesg
    (forall ((x mesg)) (implies (fact neq x x) (false)))))
